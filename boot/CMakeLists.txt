cmake_minimum_required(VERSION 3.16)
project(artemis_boot)

# Find LLVM
find_package(LLVM REQUIRED CONFIG)
message(STATUS "Found LLVM ${LLVM_PACKAGE_VERSION}")
message(STATUS "Using LLVMConfig.cmake in: ${LLVM_DIR}")

include_directories(${LLVM_INCLUDE_DIRS})
separate_arguments(LLVM_DEFINITIONS_LIST NATIVE_COMMAND ${LLVM_DEFINITIONS})
add_definitions(${LLVM_DEFINITIONS_LIST})

# Artemis compiler binary
set(ARTEMIS_EXE "${CMAKE_CURRENT_SOURCE_DIR}/../build/artemis.exe")
if(NOT EXISTS "${ARTEMIS_EXE}")
    set(ARTEMIS_EXE "${CMAKE_CURRENT_SOURCE_DIR}/../build/artemis")
endif()

# Stdlib path
set(ARTEMIS_STDLIB_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../compiler/std/include"
    CACHE PATH "Artemis standard library include path")

# The compiled output (LLVM IR generated by artemis, then assembled)
set(BOOT_ARC "${CMAKE_CURRENT_SOURCE_DIR}/main.arc")
set(BOOT_LL  "${CMAKE_CURRENT_BINARY_DIR}/artemis_boot.ll")
set(BOOT_OBJ "${CMAKE_CURRENT_BINARY_DIR}/artemis_boot.o")

# Get LLVM libraries
llvm_map_components_to_libnames(llvm_libs
    Core
    Support
    Target
    TargetParser
    MC
    X86
    X86AsmParser
    X86CodeGen
    X86Desc
    X86Info
    Passes
    Analysis
    TransformUtils
    InstCombine
    Instrumentation
    AggressiveInstCombine
    ScalarOpts
    BitWriter
    IRReader
    Linker
    Object
    AsmPrinter
    SelectionDAG
    CodeGen
    GlobalISel
)

# Step 1: Compile main.arc to LLVM IR using the existing artemis compiler
add_custom_command(
    OUTPUT "${BOOT_LL}"
    COMMAND "${ARTEMIS_EXE}"
        "${BOOT_ARC}"
        -S
        -o "${BOOT_LL}"
        -DARTEMIS_STDLIB_PATH="${ARTEMIS_STDLIB_PATH}"
    DEPENDS "${BOOT_ARC}"
    COMMENT "Compiling artemis_boot main.arc to LLVM IR"
    VERBATIM
)

# Step 2: Assemble IR to object file
add_custom_command(
    OUTPUT "${BOOT_OBJ}"
    COMMAND llc -filetype=obj -o "${BOOT_OBJ}" "${BOOT_LL}"
    DEPENDS "${BOOT_LL}"
    COMMENT "Assembling artemis_boot LLVM IR to object file"
    VERBATIM
)

# Step 3: Link with LLVM libraries to produce the self-hosting compiler
add_executable(artemis_boot_exe IMPORTED)
add_custom_target(artemis_boot ALL
    COMMAND "${CMAKE_CXX_COMPILER}"
        "${BOOT_OBJ}"
        -o "${CMAKE_CURRENT_BINARY_DIR}/artemis_boot"
        ${llvm_libs}
        -lstdc++
        -lm
    DEPENDS "${BOOT_OBJ}"
    COMMENT "Linking artemis_boot"
)
